```python
def get_user_settings(user_id: int) -> Tuple[str, int, int]:
    """
    Получает настройки пользователя (язык, начало тихих часов, конец тихих часов).
    ВСЕГДА возвращает кортеж с гарантированными значениями - НИКОГДА не возвращает None.
    
    Returns:
        Tuple[str, int, int]: (language, quiet_start_hour, quiet_end_hour)
        Defaults: ("ru", 23, 7) - русский язык, тихие часы с 23:00 до 07:00
    """
    try:
        conn = sqlite3.connect(DATABASE_PATH)
        cursor = conn.cursor()
        
        # Получаем настройки пользователя
        cursor.execute(
            "SELECT language, quiet_start, quiet_end FROM users WHERE user_id = ?",
            (user_id,)
        )
        result = cursor.fetchone()
        conn.close()
        
        # Гарантируем возврат кортежа с дефолтными значениями
        if result and result[0] and result[1] is not None and result[2] is not None:
            return result
        
        # Возвращаем дефолты если что-то отсутствует
        return ("ru", 23, 7)  # ✅ НИКОГДА не None
    except Exception as e:
        logger.error(f"Error getting user settings: {e}")
        return ("ru", 23, 7)  # ✅ Graceful fallback
```

### НОВАЯ ФУНКЦИЯ: save_price_point()

```python
def save_price_point(subscription_id: int, price: float, ts: int = None, max_points: int = 10000):
    """
    Сохраняет точку цены с дедупликацией и ограничением размера истории.
    - Не сохраняет цены которые были сохранены менее часа назад (дедупликация)
    - Ограничивает историю максимум 10000 записей
    - Если лимит превышен - удаляет самые старые записи
    """
    if ts is None:
        ts = int(time.time())
    
    try:
        conn = sqlite3.connect(DATABASE_PATH)
        cursor = conn.cursor()
        
        # Проверяем последнюю записанную цену
        cursor.execute(
            "SELECT price, timestamp FROM price_history WHERE subscription_id = ? ORDER BY timestamp DESC LIMIT 1",
            (subscription_id,)
        )
        last = cursor.fetchone()
        
        # Дедупликация: если цена записана менее часа назад и равна текущей - не записываем
        if last:
            last_price, last_ts = last
            if abs(ts - last_ts) < 3600 and abs(last_price - price) < 0.01:
                conn.close()
                return
        
        # Добавляем новую точку
        cursor.execute(
            "INSERT INTO price_history (subscription_id, price, timestamp) VALUES (?, ?, ?)",
            (subscription_id, price, ts)
        )
        
        # Проверяем размер истории
        cursor.execute(
            "SELECT COUNT(*) FROM price_history WHERE subscription_id = ?",
            (subscription_id,)
        )
        count = cursor.fetchone()[0]
        
        # Если превышен лимит - удаляем самые старые
        if count > max_points:
            excess = count - max_points
            cursor.execute(
                """DELETE FROM price_history WHERE subscription_id = ? AND timestamp IN (
                    SELECT timestamp FROM price_history WHERE subscription_id = ? 
                    ORDER BY timestamp ASC LIMIT ?
                )""",
                (subscription_id, subscription_id, excess)
            )
        
        conn.commit()
        conn.close()
    except Exception as e:
        logger.error(f"Error saving price point: {e}")
```

```python
def update_user_settings(user_id: int, **kwargs):
    """
    Безопасно обновляет настройки пользователя с параметризованными запросами.
    Поддерживаемые поля: language, quiet_start, quiet_end
    
    Example:
        update_user_settings(123, language="en", quiet_start=22)
    """
    try:
        conn = sqlite3.connect(DATABASE_PATH)
        cursor = conn.cursor()
        
        # Разрешенные поля
        allowed_fields = {"language", "quiet_start", "quiet_end"}
        
        # Фильтруем только разрешенные поля
        updates = {k: v for k, v in kwargs.items() if k in allowed_fields}
        
        if not updates:
            conn.close()
            return
        
        # Строим параметризованный запрос
        set_clause = ", ".join([f"{k} = ?" for k in updates.keys()])
        values = list(updates.values()) + [user_id]
        
        query = f"UPDATE users SET {set_clause} WHERE user_id = ?"
        cursor.execute(query, values)
        
        conn.commit()
        conn.close()
    except Exception as e:
        logger.error(f"Error updating user settings: {e}")
```
